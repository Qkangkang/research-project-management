```vue
  try {

    console.error('加载待办数量失败', error)
```

### 2.3 AppSidebar - 侧边栏菜单

**文件路径：** `src/components/Layout/AppSidebar.vue`

**职责：** 显示导航菜单，支持折叠/展开

**Props：**

```typescript
interface Props {
  collapsed?: boolean  // 是否折叠
}
```

**Emits：**

```typescript
const emit = defineEmits<{
  toggle: []  // 切换折叠状态
}>()
```

**结构：**

```vue
<template>
  <div class="app-sidebar">
    <el-menu
      :default-active="activeMenu"
      :collapse="collapsed"
      :unique-opened="true"
      router
    >
      <template v-for="route in menuRoutes" :key="route.path">
        <!-- 单级菜单 -->
        <el-menu-item 
          v-if="!route.children"
          :index="route.path"
          v-permission="route.meta?.permission"
        >
          <el-icon><component :is="route.meta?.icon" /></el-icon>
          <template #title>{{ route.meta?.title }}</template>
        </el-menu-item>
        
        <!-- 多级菜单 -->
        <el-sub-menu 
          v-else
          :index="route.path"
          v-permission="route.meta?.permission"
        >
          <template #title>
            <el-icon><component :is="route.meta?.icon" /></el-icon>
            <span>{{ route.meta?.title }}</span>
          </template>
          <el-menu-item
            v-for="child in route.children"
            :key="child.path"
            :index="child.path"
            v-permission="child.meta?.permission"
          >
            {{ child.meta?.title }}
          </el-menu-item>
        </el-sub-menu>
      </template>
    </el-menu>
    
    <div class="app-sidebar__footer">
      <el-button 
        text
        @click="emit('toggle')"
      >
        <el-icon>
          <Fold v-if="!collapsed" />
          <Expand v-else />
        </el-icon>
      </el-button>
    </div>
  </div>
</template>

<script setup lang="ts">
import { computed } from 'vue'
import { useRoute } from 'vue-router'
import { Fold, Expand } from '@element-plus/icons-vue'
import { routes } from '@/router/routes'

interface Props {
  collapsed?: boolean
}

const props = withDefaults(defineProps<Props>(), {
  collapsed: false
})

const emit = defineEmits<{
  toggle: []
}>()

const route = useRoute()

const activeMenu = computed(() => route.path)

// 过滤出需要在菜单显示的路由
const menuRoutes = computed(() => {
  return routes
    .find(r => r.path === '/')
    ?.children
    ?.filter(r => !r.meta?.hidden) || []
})
</script>
```

### 2.4 AppTabs - 标签页管理

**文件路径：** `src/components/Layout/AppTabs.vue`

**职责：** 显示已打开的页面标签，支持关闭、刷新

**Props：** 无

**结构：**

```vue
<template>
  <div class="app-tabs">
    <el-tabs
      v-model="activeTab"
      type="card"
      @tab-click="handleTabClick"
      @tab-remove="handleTabRemove"
    >
      <el-tab-pane
        v-for="tab in visitedViews"
        :key="tab.path"
        :label="tab.title"
        :name="tab.path"
        :closable="tab.path !== '/dashboard'"
      />
    </el-tabs>
    
    <el-dropdown @command="handleCommand">
      <el-button text>
        <el-icon><ArrowDown /></el-icon>
      </el-button>
      <template #dropdown>
        <el-dropdown-menu>
          <el-dropdown-item command="refresh">刷新当前</el-dropdown-item>
          <el-dropdown-item command="closeOthers">关闭其他</el-dropdown-item>
          <el-dropdown-item command="closeAll">关闭所有</el-dropdown-item>
        </el-dropdown-menu>
      </template>
    </el-dropdown>
  </div>
</template>

<script setup lang="ts">
import { computed } from 'vue'
import { useRouter, useRoute } from 'vue-router'
import { ArrowDown } from '@element-plus/icons-vue'
import { useAppStore } from '@/stores/app'

const router = useRouter()
const route = useRoute()
const appStore = useAppStore()

const activeTab = computed({
  get: () => route.path,
  set: (path: string) => {
    router.push(path)
  }
})

const visitedViews = computed(() => appStore.visitedViews)

const handleTabClick = (pane: any) => {
  router.push(pane.props.name)
}

const handleTabRemove = (path: string) => {
  appStore.closeView(path)
  // 如果关闭的是当前页，跳转到最后一个标签
  if (path === route.path) {
    const views = appStore.visitedViews
    const lastView = views[views.length - 1]
    if (lastView) {
      router.push(lastView.path)
    } else {
      router.push('/dashboard')
    }
  }
}

const handleCommand = (command: string) => {
  switch (command) {
    case 'refresh':
      router.go(0)
      break
    case 'closeOthers':
      appStore.closeOtherViews(route.path)
      break
    case 'closeAll':
      appStore.closeAllViews()
      router.push('/dashboard')
      break
  }
}
</script>

<style scoped lang="scss">
.app-tabs {
  display: flex;
  align-items: center;
  padding: 0 10px;
  background: #fff;
  box-shadow: 0 1px 4px rgba(0, 0, 0, 0.08);
  
  .el-tabs {
    flex: 1;
  }
}
</style>
```

### 3.1 DataTable - 增强表格组件

**文件路径：** `src/components/DataTable/DataTable.vue`

**职责：** 封装 el-table，统一列表页表格逻辑（分页、选择、排序）

**Props 定义：**

```typescript
interface Props {
  columns: ColumnConfig[]       // 列配置
  data: any[]                   // 数据数组
  loading?: boolean             // 加载状态
  pagination?: PaginationConfig // 分页配置
  selectable?: boolean          // 是否可选择
  sortable?: boolean            // 是否可排序
  showIndex?: boolean           // 是否显示序号
  stripe?: boolean              // 是否斑马纹
  border?: boolean              // 是否带边框
}

interface ColumnConfig {
  prop: string                  // 字段名
  label: string                 // 列标题
  width?: number | string       // 列宽度
  minWidth?: number | string    // 最小宽度
  formatter?: (row: any, column: any, value: any) => string
  align?: 'left' | 'center' | 'right'
  fixed?: 'left' | 'right'
  sortable?: boolean | 'custom'
  slot?: string                 // 自定义插槽名称
}

interface PaginationConfig {
  page: number
  pageSize: number
  total: number
  pageSizes?: number[]
}
```

**Emits 定义：**

```typescript
const emit = defineEmits<{
  'page-change': [page: number]
  'size-change': [size: number]
  'selection-change': [rows: any[]]
  'sort-change': [{ prop: string, order: 'asc' | 'desc' | null }]
  'row-click': [row: any]
}>()
```

**组件实现：**

```vue
<template>
  <div class="data-table">
    <el-table
      v-loading="loading"
      :data="data"
      :stripe="stripe"
      :border="border"
      @selection-change="handleSelectionChange"
      @sort-change="handleSortChange"
      @row-click="handleRowClick"
    >
      <!-- 选择列 -->
      <el-table-column
        v-if="selectable"
        type="selection"
        width="55"
        align="center"
        fixed="left"
      />
      
      <!-- 序号列 -->
      <el-table-column
        v-if="showIndex"
        type="index"
        label="序号"
        width="70"
        align="center"
        :index="indexMethod"
      />
      
      <!-- 数据列 -->
      <el-table-column
        v-for="column in columns"
        :key="column.prop"
        :prop="column.prop"
        :label="column.label"
        :width="column.width"
        :min-width="column.minWidth"
        :align="column.align || 'left'"
        :fixed="column.fixed"
        :sortable="column.sortable"
        :formatter="column.formatter"
      >
        <template v-if="column.slot" #default="scope">
          <slot :name="column.slot" :row="scope.row" :column="scope.column" />
        </template>
      </el-table-column>
    </el-table>
    
    <!-- 分页 -->
    <el-pagination
      v-if="pagination"
      v-model:current-page="pagination.page"
      v-model:page-size="pagination.pageSize"
      :total="pagination.total"
      :page-sizes="pagination.pageSizes || [10, 20, 50, 100]"
      :layout="paginationLayout"
      :background="true"
      class="data-table__pagination"
      @current-change="emit('page-change', $event)"
      @size-change="emit('size-change', $event)"
    />
  </div>
</template>

<script setup lang="ts">
import { computed } from 'vue'

interface Props {
  columns: ColumnConfig[]
  data: any[]
  loading?: boolean
  pagination?: PaginationConfig
  selectable?: boolean
  sortable?: boolean
  showIndex?: boolean
  stripe?: boolean
  border?: boolean
}

interface ColumnConfig {
  prop: string
  label: string
  width?: number | string
  minWidth?: number | string
  formatter?: (row: any, column: any, value: any) => string
  align?: 'left' | 'center' | 'right'
  fixed?: 'left' | 'right'
  sortable?: boolean | 'custom'
  slot?: string
}

interface PaginationConfig {
  page: number
  pageSize: number
  total: number
  pageSizes?: number[]
}

const props = withDefaults(defineProps<Props>(), {
  loading: false,
  selectable: false,
  sortable: false,
  showIndex: false,
  stripe: true,
  border: true
})

const emit = defineEmits<{
  'page-change': [page: number]
  'size-change': [size: number]
  'selection-change': [rows: any[]]
  'sort-change': [{ prop: string, order: 'asc' | 'desc' | null }]
  'row-click': [row: any]
}>()

const paginationLayout = computed(() => {
  return 'total, sizes, prev, pager, next, jumper'
})

const indexMethod = (index: number) => {
  if (props.pagination) {
    return (props.pagination.page - 1) * props.pagination.pageSize + index + 1
  }
  return index + 1
}

const handleSelectionChange = (rows: any[]) => {
  emit('selection-change', rows)
}

const handleSortChange = ({ prop, order }: any) => {
  emit('sort-change', { prop, order })
}

const handleRowClick = (row: any) => {
  emit('row-click', row)
}
</script>

<style scoped lang="scss">
.data-table {
  &__pagination {
    margin-top: 20px;
    display: flex;
    justify-content: flex-end;
  }
}
</style>
```

```vue
  if (!formRef.value) return false
  return await formRef.value.validate()
}

const resetFields = () => {

}

defineExpose({
  validate,
  resetFields
})

  initFormData()
}, { deep: true })

onMounted(() => {
  initFormData()
})
</script>
```

### 4.2 FileUpload - 文件上传组件

**文件路径：** `src/components/FileUpload/FileUpload.vue`

**职责：** 文件上传，支持多文件、类型限制、大小限制

**Props 定义：**

```typescript
interface Props {
```

**Emits：**

```typescript
const emit = defineEmits<{
  'update:modelValue': [value: string[]]
  success: [fileId: string]
  error: [error: Error]

```

**组件实现：**

```vue
<template>
  <div class="file-upload">
    <el-upload
      ref="uploadRef"
      :action="uploadAction"
      :headers="uploadHeaders"
      :accept="accept"
      :before-upload="beforeUpload"
      :on-success="handleSuccess"
      :on-error="handleError"
      :on-remove="handleRemove"
      :file-list="fileList"
      :list-type="listType"
      :disabled="disabled"
      :limit="maxCount"
      :on-exceed="handleExceed"
    >
      <template v-if="listType === 'picture-card'">
        <el-icon><Plus /></el-icon>
      </template>
      <template v-else>
        <el-button :disabled="disabled">
          <el-icon><Upload /></el-icon>
          点击上传
        </el-button>
      </template>
      <template #tip>
        <div class="el-upload__tip">
          <span v-if="accept">支持格式：{{ accept }}</span>
          <span v-if="maxSize">，单个文件不超过 {{ maxSize }}MB</span>
          <span v-if="maxCount">，最多上传 {{ maxCount }} 个文件</span>
        </div>
      </template>
    </el-upload>
  </div>
</template>

<script setup lang="ts">
import { ref, computed, watch } from 'vue'
import { ElMessage } from 'element-plus'
import { Plus, Upload } from '@element-plus/icons-vue'
import type { UploadProps, UploadUserFile } from 'element-plus'
import { getToken } from '@/utils/auth'
import { fileApi } from '@/api/file'

interface Props {
  modelValue: string[]
  accept?: string
  maxSize?: number
  maxCount?: number
  disabled?: boolean
  listType?: 'text' | 'picture' | 'picture-card'
}

const props = withDefaults(defineProps<Props>(), {
  accept: '*',
  maxSize: 50,
  maxCount: 5,
  disabled: false,
  listType: 'text'
})

const emit = defineEmits<{
  'update:modelValue': [value: string[]]
  success: [fileId: string]
  error: [error: Error]
}>()

const uploadRef = ref()
const fileList = ref<UploadUserFile[]>([])

const uploadAction = computed(() => {
  return `${import.meta.env.VITE_API_BASE_URL}/files/upload`
})

const uploadHeaders = computed(() => {
  return {
    Authorization: `Bearer ${getToken()}`
  }
})

const beforeUpload: UploadProps['beforeUpload'] = (rawFile) => {
  // 文件大小校验
  if (props.maxSize && rawFile.size / 1024 / 1024 > props.maxSize) {
    ElMessage.error(`文件大小不能超过 ${props.maxSize}MB`)
    return false
  }
  
  // 文件类型校验
  if (props.accept && props.accept !== '*') {
    const acceptTypes = props.accept.split(',').map(t => t.trim())
    const fileExtension = '.' + rawFile.name.split('.').pop()?.toLowerCase()
    if (!acceptTypes.includes(fileExtension)) {
      ElMessage.error(`只支持上传 ${props.accept} 格式的文件`)
      return false
    }
  }
  
  return true
}

const handleSuccess: UploadProps['onSuccess'] = (response, uploadFile) => {
  if (response.code === 'OK') {
    const fileId = response.data.fileId
    const newFileIds = [...props.modelValue, fileId]
    emit('update:modelValue', newFileIds)
    emit('success', fileId)
    
    ElMessage.success('上传成功')
  } else {
    ElMessage.error(response.message || '上传失败')
    // 移除失败的文件
    uploadRef.value?.handleRemove(uploadFile)
  }
}

const handleError: UploadProps['onError'] = (error) => {
  ElMessage.error('上传失败：' + error.message)
  emit('error', error)
}

const handleRemove: UploadProps['onRemove'] = (uploadFile) => {
  const fileId = uploadFile.response?.data?.fileId || uploadFile.uid
  const newFileIds = props.modelValue.filter(id => id !== fileId)
  emit('update:modelValue', newFileIds)
}

const handleExceed: UploadProps['onExceed'] = () => {
  ElMessage.warning(`最多只能上传 ${props.maxCount} 个文件`)
}

// 加载已上传的文件列表
const loadFileList = async () => {
  if (props.modelValue.length === 0) {
    fileList.value = []
    return
  }
  
  try {
    const files = await Promise.all(
      props.modelValue.map(id => fileApi.getFileInfo(id))
    )
    fileList.value = files.map(file => ({
      uid: file.id,
      name: file.fileName,
      url: file.downloadUrl,
      response: {
        code: 'OK',
        data: { fileId: file.id }
      }
    }))
  } catch (error) {
    console.error('加载文件列表失败', error)
  }
}

watch(() => props.modelValue, () => {
  loadFileList()
}, { immediate: true })
</script>

<style scoped lang="scss">
.file-upload {
  :deep(.el-upload__tip) {
    color: #909399;
    font-size: 12px;
    margin-top: 8px;
  }
}
</style>
```

```vue
<template>
  <div class="budget-editor">
    <el-button 
      v-if="!disabled"
      type="primary" 
      size="small"
      @click="handleAddLine"
    >
      <el-icon><Plus /></el-icon>
      添加科目
    </el-button>
    
    <el-table 
      :data="lines" 
      border
      show-summary
      :summary-method="getSummaries"
      class="budget-editor__table"
    >
      <el-table-column prop="subjectName" label="预算科目" width="200">
        <template #default="{ row, $index }">
          <el-select
            v-model="row.subjectCode"
            :disabled="disabled"
            placeholder="请选择科目"
            @change="handleSubjectChange($index)"
          >
            <el-option
              v-for="subject in subjectOptions"
              :key="subject.code"
              :label="subject.name"
              :value="subject.code"
              :disabled="isSubjectUsed(subject.code, $index)"
            />
          </el-select>
        </template>
      </el-table-column>
      
      <el-table-column prop="amount" label="预算金额（元）" width="200">
        <template #default="{ row }">
          <el-input-number
            v-model="row.amount"
            :disabled="disabled"
            :min="0"
            :precision="2"
            :controls="false"
            style="width: 100%"
            @change="handleAmountChange"
          />
        </template>
      </el-table-column>
      
      <el-table-column label="占比" width="100">
        <template #default="{ row }">
          {{ ((row.amount / totalAmount) * 100).toFixed(2) }}%
        </template>
      </el-table-column>
      
      <el-table-column 
        v-if="!disabled"
        label="操作" 
        width="80" 
        fixed="right"
      >
        <template #default="{ $index }">
          <el-button
            link
            type="danger"
            @click="handleRemoveLine($index)"
          >
            删除
          </el-button>
        </template>
      </el-table-column>
    </el-table>
    
    <div class="budget-editor__total">
      <span>预算总额：</span>
      <span class="budget-editor__total-amount">
        {{ formatMoney(totalAmount) }} 元
      </span>
    </div>
  </div>
</template>

<script setup lang="ts">
import { ref, computed, watch } from 'vue'
import { Plus } from '@element-plus/icons-vue'
import { ElMessage } from 'element-plus'
import { formatMoney } from '@/utils/number'

interface Props {
  modelValue: BudgetLine[]
  disabled?: boolean
  subjectOptions: SubjectOption[]
}

interface BudgetLine {
  subjectCode: string
  subjectName: string
  amount: number
}

interface SubjectOption {
  code: string
  name: string
}

const props = withDefaults(defineProps<Props>(), {
  disabled: false
})

const emit = defineEmits<{
  'update:modelValue': [value: BudgetLine[]]
  'total-change': [total: number]
}>()

const lines = ref<BudgetLine[]>([])

const totalAmount = computed(() => {
  return lines.value.reduce((sum, line) => sum + (line.amount || 0), 0)
})

const handleAddLine = () => {
  lines.value.push({
    subjectCode: '',
    subjectName: '',
    amount: 0
  })
  emitUpdate()
}

const handleRemoveLine = (index: number) => {
  lines.value.splice(index, 1)
  emitUpdate()
}

const handleSubjectChange = (index: number) => {
  const line = lines.value[index]
  const subject = props.subjectOptions.find(s => s.code === line.subjectCode)
  if (subject) {
    line.subjectName = subject.name
    emitUpdate()
  }
}

const handleAmountChange = () => {
  emitUpdate()
}

const isSubjectUsed = (code: string, currentIndex: number) => {
  return lines.value.some((line, index) => 
    index !== currentIndex && line.subjectCode === code
  )
}

const getSummaries = (param: any) => {
  const { columns } = param
  const sums: string[] = []
  columns.forEach((column: any, index: number) => {
    if (index === 0) {
      sums[index] = '总计'
    } else if (column.property === 'amount') {
      sums[index] = formatMoney(totalAmount.value)
    } else {
      sums[index] = ''
    }
  })
  return sums
}

const emitUpdate = () => {
  emit('update:modelValue', lines.value)
  emit('total-change', totalAmount.value)
}

watch(() => props.modelValue, (newValue) => {
  lines.value = JSON.parse(JSON.stringify(newValue || []))
}, { immediate: true, deep: true })

watch(totalAmount, (newTotal) => {
  emit('total-change', newTotal)
})
</script>

    margin-top: 16px;
  }
  
  &__total {
    margin-top: 16px;
    text-align: right;
    font-size: 16px;
    font-weight: 500;
  }
  
  &__total-amount {
    color: #409eff;
    font-size: 20px;
    margin-left: 8px;
  }
}
</style>
```

**组件实现：**

```vue
<template>
  <el-button
    v-if="hasPermission"
    :type="type"
    :size="size"
    :disabled="disabled"
    :loading="loading"
    v-bind="$attrs"
    @click="handleClick"
  >
    <slot />
  </el-button>
</template>

<script setup lang="ts">
import { computed } from 'vue'
import { usePermission } from '@/composables/usePermission'

interface Props {
  permission: string | string[]
  type?: any
  size?: any
  disabled?: boolean
  loading?: boolean
}

const props = withDefaults(defineProps<Props>(), {
  type: 'default',
  size: 'default',
  disabled: false,
  loading: false
})

const emit = defineEmits<{
  click: [event: MouseEvent]
}>()

const { hasPermission: checkPermission, hasAnyPermission } = usePermission()

const hasPermission = computed(() => {
  if (Array.isArray(props.permission)) {
    return hasAnyPermission(...props.permission)
  }
  return checkPermission(props.permission)
})

const handleClick = (event: MouseEvent) => {
  emit('click', event)
}
</script>
```

### 6.1 ProjectSelector - 项目选择器

**文件路径：** `src/components/ProjectSelector/ProjectSelector.vue`

**职责：** 项目选择下拉框，支持搜索

**Props 定义：**

```typescript
interface Props {
  modelValue: string | string[]  // 项目ID
  multiple?: boolean             // 是否多选
  disabled?: boolean             // 是否禁用
  placeholder?: string           // 占位符
  filterStatus?: string[]        // 过滤项目状态
}
```

**Emits：**

```typescript
const emit = defineEmits<{
  'update:modelValue': [value: string | string[]]
  change: [value: string | string[]]
}>()
```

**组件实现：**

```vue
<template>
  <el-select
    :model-value="modelValue"
    :multiple="multiple"
    :disabled="disabled"
    :placeholder="placeholder"
    filterable
    remote
    :remote-method="handleSearch"
    :loading="loading"
    @update:model-value="handleChange"
  >
    <el-option
      v-for="project in projects"
      :key="project.id"
      :label="`${project.projectNo} - ${project.projectName}`"
      :value="project.id"
    />
  </el-select>
</template>

<script setup lang="ts">
import { ref, onMounted } from 'vue'
import { projectApi } from '@/api/project'
import type { Project } from '@/api/project'

interface Props {
  modelValue: string | string[]
  multiple?: boolean
  disabled?: boolean
  placeholder?: string
  filterStatus?: string[]
}

const props = withDefaults(defineProps<Props>(), {
  multiple: false,
  disabled: false,
  placeholder: '请选择项目'
})

const emit = defineEmits<{
  'update:modelValue': [value: string | string[]]
  change: [value: string | string[]]
}>()

const loading = ref(false)
const projects = ref<Project[]>([])

const loadProjects = async (keyword?: string) => {
  loading.value = true
  try {
    const res = await projectApi.list({
      keyword,
      status: props.filterStatus,
      page: 1,
      pageSize: 20
    })
    projects.value = res.items
  } finally {
    loading.value = false
  }
}

const handleSearch = (query: string) => {
  if (query) {
    loadProjects(query)
  } else {
    loadProjects()
  }
}

const handleChange = (value: string | string[]) => {
  emit('update:modelValue', value)
  emit('change', value)
}

onMounted(() => {
  loadProjects()
})
</script>
```
